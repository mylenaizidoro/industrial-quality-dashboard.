function montarGrafico(meses, valores, objetivos) { const ctx = document.getElementById("graficoAbsenteismo"); const coresBarras = valores.map((valor, i) => { return valor <= objetivos[i] ? "#10b981" : "#f43f5e"; }); new Chart(ctx, { type: "bar", data: { labels: meses, datasets: [ { label: "Realizado", data: valores, backgroundColor: coresBarras, borderRadius: 8, barThickness: 22 }, { label: "Meta", data: objetivos, type: "line", borderColor: "#ff7a00", backgroundColor: "#ff7a00", borderWidth: 2, borderDash: [6, 6], pointRadius: 4, pointBackgroundColor: "#ffffff", pointBorderColor: "#ff7a00", pointBorderWidth: 2, tension: 0 } ] }, options: { responsive: true, maintainAspectRatio: false, plugins: { title: { display: true, text: "ABSENTEÍSMO PA - 03", align: "start", color: "#0b1f3a", font: { size: 24, weight: "bold" } }, legend: { position: "top", align: "center" }, tooltip: { callbacks: { label: function(context) { return context.dataset.label + ": " + context.raw.toFixed(2) + "%"; } } } }, scales: { y: { beginAtZero: true, max: 9, ticks: { stepSize: 1 }, grid: { color: "rgba(0,0,0,0.10)" } }, x: { ticks: { maxRotation: 60, minRotation: 60 }, grid: { color: "rgba(0,0,0,0.08)" } } } }, plugins: [{ afterDatasetsDraw(chart) { const { ctx } = chart; chart.data.datasets[0].data.forEach((valor, index) => { const meta = chart.getDatasetMeta(0); const barra = meta.data[index]; ctx.save(); ctx.fillStyle = "#111"; ctx.font = "bold 11px Arial"; ctx.textAlign = "center"; ctx.translate(barra.x, barra.y - 8); ctx.rotate(-Math.PI / 2); ctx.fillText(valor.toFixed(2).replace(".", ",") + "%", 0, 0); ctx.restore(); }); } }] }); }
